//go:build integration && auditcrash
package integrationtest
import (
 "os"
 "testing"
 "github.com/stretchr/testify/require"
 "github.com/testcontainers/testcontainers-go"
)
func TestAuditCrashCleanup(t *testing.T) {
 c := Start(t, testcontainers.ContainerRequest{Image:"postgres:18.6-alpine3.24@sha256:10a5c69a83035ab5239f5620d708520e8b28f5aa1f2b20143e6024532edadfb5", ExposedPorts:[]string{"5432/tcp"}, Env:map[string]string{"POSTGRES_PASSWORD":"audit-only"}})
 require.NoError(t, os.WriteFile("/tmp/DEPS-loop2/crash-container-id", []byte(c.GetContainerID()), 0600))
 os.Exit(23)
}
